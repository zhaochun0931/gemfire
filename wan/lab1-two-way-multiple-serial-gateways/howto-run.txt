Configure Cluster 1 (System ID 1)



1.Configure Cluster 1 (System ID 1):Set up receivers, senders targeting ID 2, and regions.


docker exec -it cluster1-locator gfsh

connect --locator=cluster1-locator[10334]


Create Gateway Receiver:

create gateway-receiver --start-port=50504 --end-port=50510




Create 3 Serial Gateway Senders targeting Distributed System ID 2:

create gateway-sender --id=Sender1ToCluster2 --remote-distributed-system-id=2 --parallel=false --enable-persistence=true

create gateway-sender --id=Sender2ToCluster2 --remote-distributed-system-id=2 --parallel=false --enable-persistence=true

create gateway-sender --id=Sender3ToCluster2 --remote-distributed-system-id=2 --parallel=false --enable-persistence=true



Create 5 Partitioned Regions attached to all 3 Gateway Senders:

create region --name=OrderData --type=PARTITION_REDUNDANT --gateway-sender-id=Sender1ToCluster2,Sender2ToCluster2,Sender3ToCluster2

create region --name=CustomerData --type=PARTITION_REDUNDANT --gateway-sender-id=Sender1ToCluster2,Sender2ToCluster2,Sender3ToCluster2

create region --name=ProductData --type=PARTITION_REDUNDANT --gateway-sender-id=Sender1ToCluster2,Sender2ToCluster2,Sender3ToCluster2

create region --name=InventoryData --type=PARTITION_REDUNDANT --gateway-sender-id=Sender1ToCluster2,Sender2ToCluster2,Sender3ToCluster2

create region --name=ShippingData --type=PARTITION_REDUNDANT --gateway-sender-id=Sender1ToCluster2,Sender2ToCluster2,Sender3ToCluster2
